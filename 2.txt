    ввести ЭлектроннуюПочту
    ввести Пароль
    ввести ПовторныйПароль

    если Пароль не равен ПовторныйПароль
        вывести "Введите пароль еще раз"
        обнулить Пароль и ПовторныйПароль
        повторить процедура Регистрация()
    иначе
        id = получитьСледующийID()
        запись = {
            ID: id,
            Имя: Имя,
            Фамилия: Фамилия,
            ЭлектроннаяПочта: ЭлектроннаяПочта,
            Пароль: Пароль
        }
        вставитьВТаблицу("ТАБПОЛЬ", запись)
        перейтиКГлавномуМеню()
    конечное если
конец процедуры

процедура Вход()
    инициализация попыток = 0
    повторять
        ввести ЭлектроннуюПочту
        ввести Пароль

        пользователь = найтиВТаблице("ТАБПОЛЬ", ЭлектроннаяПочта)
        если пользователь не найден
            вывести "Неправильный пользователь"
        иначе
            если Пароль не равен пользователь.Пароль
                вывести "Неправильный пароль"
                попытки = попытки + 1
            иначе
                перейтиКГлавномуМеню()
                выйти из цикла
            конец если
        конечное если

        если попытки >= 5
            вывести "Слишком много попыток!"
            заблокироватьУстройство(15 минут)
            выйти из цикла
        конец если
    пока попытки < 5
конец процедуры
